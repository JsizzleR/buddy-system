func TestVerificationBrokenStdoutStillClosesWorker(t *testing.T){
 f:=newEntryFixture(t,"")
 cmd:=exec.Command(f.adapter,"--harness","codex","--executable",f.codex,"--buddy",f.buddy,"--launch",f.launch,"--receipt",f.receipt,"--label","worker")
 cmd.Env=f.env;input,err:=cmd.StdinPipe();if err!=nil{t.Fatal(err)};defer input.Close()
 output,err:=cmd.StdoutPipe();if err!=nil{t.Fatal(err)};defer output.Close();stderr:=&lockedBuffer{};cmd.Stderr=stderr;cmd.WaitDelay=2*time.Second
 if err=cmd.Start();err!=nil{t.Fatal(err)};defer cmd.Process.Kill()
 var start map[string]any;if err=json.NewDecoder(output).Decode(&start);err!=nil||start["operation"]!="start"{t.Fatalf("start control: %v %v",start,err)}
 var leader,child int
 for _,row:=range nativeLog(t,f.marker){if row["kind"]=="app-server"{leader=int(row["pid"].(float64));child=int(row["tool"].(float64))}}
 if leader==0||child==0{t.Fatal("fixture processes absent")};defer syscall.Kill(leader,syscall.SIGKILL);defer syscall.Kill(child,syscall.SIGKILL)
 output.Close();f.release(t)
 wait:=make(chan error,1);go func(){wait<-cmd.Wait()}()
 select{case <-wait:case <-time.After(10*time.Second):t.Fatal("broken output did not stop adapter")}
 if !strings.Contains(stderr.String(),"control output unavailable"){t.Fatalf("output error bypassed orderly shutdown: %s",stderr.String())}
 if !pidGone(leader,2*time.Second)||!pidGone(child,2*time.Second){t.Fatal("broken output orphaned native process tree")}
}
