func TestVerificationLegacyJournalAndContradictoryRefusal(t *testing.T){
 for _,state:=range []string{"unknown","accepted","delivered","refused-valid","refused-prior-turn","refused-own-turn"}{
 t.Run(state,func(t *testing.T){
  id:=Identity{"buddy.lifecycle.v1","codex","native-1","captured"};path:=filepath.Join(t.TempDir(),"receipt.json")
  rows:=[]Receipt{{Schema:Schema,Identity:id,Operation:"start"}}
  d:=Delivery{State:state};if state!="unknown" {d.TurnID="turn-known"}
  if state=="refused-valid"||state=="refused-prior-turn"||state=="refused-own-turn" {d=Delivery{State:"unknown"};if state=="refused-prior-turn" {d.TurnID="turn-known"}}
  rows=append(rows,Receipt{Schema:Schema,Identity:id,Operation:"send",MessageID:42,Delivery:&d})
  if state=="refused-valid"||state=="refused-prior-turn"||state=="refused-own-turn" {r:=Delivery{State:"not_submitted"};if state=="refused-own-turn" {r.TurnID="turn-already-created"};rows=append(rows,Receipt{Schema:Schema,Identity:id,Operation:"send",MessageID:42,Delivery:&r})}
  f,err:=os.Create(path+".events");if err!=nil {t.Fatal(err)};for _,row:=range rows {if err=json.NewEncoder(f).Encode(row);err!=nil {t.Fatal(err)}};f.Close()
  c,err:=NewController(&fakeBackend{},&fakeLedger{identity:id},path)
  if state=="refused-prior-turn"||state=="refused-own-turn" {if err==nil {c.Close();t.Fatal("contradictory refusal discarded reservation")};return}
  if err!=nil {t.Fatal(err)};defer c.Close()
  _,held:=c.sent[42];if state=="refused-valid" {if held {t.Fatal("valid refusal kept reservation")}}else if !held || c.sent[42]!=d {t.Fatal("legacy journal reservation changed")}
 })}
}
